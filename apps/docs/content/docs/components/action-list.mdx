<ActionListPreview />

Type in the search bar to filter items. Click an item to select it.

## Usage

```tsx
import { useState } from 'react';
import ActionList from '@/components/ui/ActionList';

const items = [
  { id: 1, displayName: 'Apple' },
  { id: 2, displayName: 'Banana' },
  { id: 3, displayName: 'Cherry' },
];

function Picker() {
  const [selected, setSelected] = useState<string | number | undefined>(undefined);

  return (
    <ActionList
      items={items}
      selectedItem={selected}
      onSelect={setSelected}
      placeholder="Search..."
      noMatchingResultsText="No matching items"
      width="240px"
      listHeight={200}
    />
  );
}
```

## Without Search Bar

```tsx
<ActionList
  items={items}
  hasSearchBar={false}
  onSelect={setSelected}
  noMatchingResultsText="No items"
/>
```

## Option Type

```ts
type Option = {
  id: string | number;
  displayName: string;
  disabled?: boolean;
};
```

## Props

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `items` | `Option[]` | — | **Required.** List of selectable items |
| `noMatchingResultsText` | `string` | — | **Required.** Empty state message |
| `onSelect` | `(value: string \| number) => void` | — | Called when an item is clicked |
| `selectedItem` | `string \| number` | — | Currently selected item ID |
| `hasSearchBar` | `boolean` | `true` | Show/hide the search bar |
| `placeholder` | `string` | `''` | Search bar placeholder |
| `onSearchTermChange` | `(value: string) => void` | — | Called on each search keystroke |
| `onSearch` | `(value: string) => void` | — | Called when search is submitted |
| `width` | `string` | `'100%'` | Width of the list container |
| `listHeight` | `number` | — | Fixed height (px) for the scrollable list |
| `className` | `string` | — | Additional CSS classes |
| `style` | `CSSProperties` | — | Inline styles |
