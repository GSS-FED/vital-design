## Usage

```tsx
import { useState } from 'react';
import Checkbox from '@/components/ui/Checkbox';

function Agreement() {
  const [checked, setChecked] = useState(false);

  return (
    <Checkbox checked={checked} onChange={setChecked}>
      I accept the terms and conditions
    </Checkbox>
  );
}
```

The `indeterminate` state is useful for "select all" patterns where only some items are selected.

## Indeterminate Pattern

```tsx
const allChecked = items.every((i) => i.checked);
const someChecked = items.some((i) => i.checked);

<Checkbox
  checked={allChecked ? true : someChecked ? 'indeterminate' : false}
  onChange={(checked) => setAll(checked)}
>
  Select all
</Checkbox>
```

## Props

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `checked` | `boolean \| 'indeterminate'` | — | **Required.** Checkbox state |
| `onChange` | `(checked: boolean) => void` | — | **Required.** Called on change (always passes `boolean`, never `'indeterminate'`) |
| `children` | `ReactNode` | — | Label text |
| `disabled` | `boolean` | `false` | Disables interaction |
| `required` | `boolean` | `false` | Applies error border when unchecked |
| `className` | `string` | — | Additional CSS classes |
| `style` | `CSSProperties` | — | Inline styles |
