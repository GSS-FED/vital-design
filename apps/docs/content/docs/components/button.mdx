## Usage

```tsx
import { Button } from '@/components/ui/Button';

<Button theme="primary" size="lg">
  Click me
</Button>;
```

## Variants

### Default

The `default` variant renders a solid background button. It is the
default rendering style.

<ComponentPreview>
  <Button variant="default" theme="primary">
    Primary
  </Button>
  <Button variant="default" theme="default">
    Default
  </Button>
  <Button variant="default" theme="success">
    Success
  </Button>
  <Button variant="default" theme="info">
    Info
  </Button>
  <Button variant="default" theme="warning">
    Warning
  </Button>
  <Button variant="default" theme="alarm">
    Alarm
  </Button>
  <Button variant="default" theme="dangerous">
    Dangerous
  </Button>
</ComponentPreview>

```tsx
<Button variant="default" theme="primary">Primary</Button>
<Button variant="default" theme="default">Default</Button>
<Button variant="default" theme="success">Success</Button>
<Button variant="default" theme="info">Info</Button>
<Button variant="default" theme="warning">Warning</Button>
<Button variant="default" theme="alarm">Alarm</Button>
<Button variant="default" theme="dangerous">Dangerous</Button>
```

### Text

The `text` variant renders a transparent background button. Available for all themes except `dangerous`.

<ComponentPreview>
  <Button variant="text" theme="primary">
    Primary
  </Button>
  <Button variant="text" theme="default">
    Default
  </Button>
  <Button variant="text" theme="success">
    Success
  </Button>
  <Button variant="text" theme="info">
    Info
  </Button>
  <Button variant="text" theme="warning">
    Warning
  </Button>
  <Button variant="text" theme="alarm">
    Alarm
  </Button>
</ComponentPreview>

```tsx
<Button variant="text" theme="primary">Primary</Button>
<Button variant="text" theme="default">Default</Button>
```

### Ghost

The `ghost` variant renders a transparent button that surfaces a
background on hover. Use it inside popovers, toolbars, and split-button
menus where the surrounding chrome provides the affordance.

<ComponentPreview>
  <Button variant="ghost" theme="primary">
    Primary
  </Button>
  <Button variant="ghost" theme="default">
    Default
  </Button>
</ComponentPreview>

```tsx
<Button variant="ghost" theme="default">Default</Button>
```

### Icons

Use `data-icon="inline-start"` or `data-icon="inline-end"` on child icons. For icon-only buttons, use `size="icon-md"` or `size="icon-lg"`.
Install or import the icon or spinner separately from the button.

<ComponentPreview>
  <Button theme="primary" size="lg">
    <SearchIcon data-icon="inline-start" />
    Search
  </Button>
  <Button theme="default" size="lg">
    Search
    <SearchIcon data-icon="inline-end" />
  </Button>
</ComponentPreview>

```tsx
<Button theme="primary">
  <SearchIcon data-icon="inline-start" />
  Search
</Button>
<Button theme="default">
  Search
  <SearchIcon data-icon="inline-end" />
</Button>
```

## Props

### Button

| Prop                    | Type                                                                                   | Default                                      | Description                                                      |
| ----------------------- | -------------------------------------------------------------------------------------- | -------------------------------------------- | ---------------------------------------------------------------- |
| `variant`               | `'default' \| 'text' \| 'ghost'`                                                       | `'default'`                                   | Visual variant                                                   |
| `theme`                 | `'primary' \| 'default' \| 'success' \| 'info' \| 'warning' \| 'alarm' \| 'dangerous'` | `'primary'` for default, `'default'` for text | Color theme (`'dangerous'` only available on `default`)          |
| `size`                  | `'md' \| 'lg' \| 'icon-md' \| 'icon-lg'`                                                 | `'md'`                                       | Button size                                                      |
| `focusableWhenDisabled` | `boolean`                                                                              | `false`                                      | Keeps the button focusable even when disabled styling is applied |
| `disabled`              | `boolean`                                                                              | `false`                                      | Native disabled attribute                                        |
| `className`             | `string`                                                                               | —                                            | Additional CSS classes                                           |
| `style`                 | `CSSProperties`                                                                        | —                                            | Inline styles                                                    |
