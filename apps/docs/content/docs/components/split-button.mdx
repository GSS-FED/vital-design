## Usage

```tsx
import { useState } from 'react';
import SplitButton from '@/components/ui/SplitButton';

function SaveButton() {
  const [open, setOpen] = useState(false);

  return (
    <SplitButton
      theme="primary"
      open={open}
      onClick={() => handleSave()}
      splitOnClick={() => setOpen((v) => !v)}
    >
      Save
    </SplitButton>
  );
}
```

## Props

| Prop | Type | Default | Description |
|------|------|---------|-------------|
| `open` | `boolean` | — | **Required.** Controls whether the split arrow appears in open state |
| `theme` | `'primary' \| 'default'` | `'primary'` | Color theme |
| `size` | `'medium' \| 'large'` | `'medium'` | Button size |
| `onClick` | `(e: MouseEvent) => void` | — | Handler for the main button area |
| `splitOnClick` | `(e: MouseEvent) => void` | — | Handler for the arrow trigger |
| `icon` | `ReactNode` | — | Icon rendered in the main button area |
| `focusableWhenDisabled` | `boolean` | `false` | Keeps keyboard focus on the button when disabled styling is applied |
| `disabled` | `boolean` | `false` | Disables the entire button |
