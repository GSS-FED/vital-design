## Usage

```tsx
import RadioGroup from '@/components/ui/RadioGroup';
import { useState } from 'react';

const options = [
  { label: 'Option A', value: 'a' },
  { label: 'Option B', value: 'b' },
  { label: 'Option C', value: 'c' },
];

function Form() {
  const [value, setValue] = useState('a');

  return (
    <RadioGroup
      options={options}
      value={value}
      onChange={setValue}
      direction="horizontal"
    />
  );
}
```

## Compound Items

Use `RadioGroupItem` when an option needs custom layout or extra text.

```tsx
import RadioGroup, {
  RadioGroupItem,
} from '@/components/ui/RadioGroup';

<RadioGroup value={value} onChange={setValue}>
  <RadioGroupItem value="basic">
    <span>Basic</span>
  </RadioGroupItem>
  <RadioGroupItem value="pro">
    <div>
      <div>Pro</div>
      <div className="text-grayscale-500">Recommended for teams</div>
    </div>
  </RadioGroupItem>
</RadioGroup>;
```

## Props

### RadioGroup

| Prop          | Type                         | Default        | Description                                 |
| ------------- | ---------------------------- | -------------- | ------------------------------------------- |
| `options`     | `RadioOption[]`              | —              | **Required.** List of radio options         |
| `value`       | `string`                     | —              | **Required.** Currently selected value      |
| `onChange`    | `(value: string) => void`    | —              | **Required.** Called when selection changes |
| `direction`   | `'horizontal' \| 'vertical'` | `'horizontal'` | Layout direction                            |
| `allowCancel` | `boolean`                    | `false`        | Clicking selected option deselects it       |
| `className`   | `string`                     | —              | Additional CSS classes                      |
| `style`       | `CSSProperties`              | —              | Inline styles                               |

### RadioGroupItem

| Prop       | Type        | Default | Description                 |
| ---------- | ----------- | ------- | --------------------------- |
| `value`    | `string`    | —       | **Required.** Option value  |
| `disabled` | `boolean`   | `false` | Disables this specific item |
| `children` | `ReactNode` | —       | Custom label content        |

### RadioOption

| Prop       | Type        | Default | Description                   |
| ---------- | ----------- | ------- | ----------------------------- |
| `label`    | `ReactNode` | —       | **Required.** Option label    |
| `value`    | `string`    | —       | **Required.** Option value    |
| `disabled` | `boolean`   | `false` | Disables this specific option |
